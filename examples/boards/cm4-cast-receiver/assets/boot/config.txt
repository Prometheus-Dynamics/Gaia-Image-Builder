# Raspberry Pi CM4 cast receiver.
[all]
arm_64bit=1
kernel=kernel8.img
disable_splash=1
boot_delay=0
initial_turbo=30
# KMS on HDMI0 (with HDMI audio); 256 MiB CMA for decoded frames. The
# BCM2711 variant by name (what overlay_map.dtb maps vc4-kms-v3d to here).
dtoverlay=vc4-kms-v3d-pi4,cma-256
max_framebuffers=1
disable_overscan=1
dtparam=audio=off
camera_auto_detect=0
display_auto_detect=0
# Bluetooth off; the PL011 UART is the serial console on GPIO14/15.
dtoverlay=disable-bt
enable_uart=1
